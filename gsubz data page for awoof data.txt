    // Ensure the script runs after the DOM is loaded
    let planList = {};

    function changeCategory(categoryValue) {
        const selectedCategory = categoryValue || category.value;
        if (!planList[selectedCategory]) {
            return;
        }
        // Build the plan options with the additional data-description attribute
        const planOptions = planList[selectedCategory].map(plan => 
            `<option value="${plan.value}" 
                     data-max="${plan.max}" 
                     data-min="${plan.min}" 
                     data-price="${plan.price}" 
                     data-description="${plan.description}">${plan.display_name} - ₦${plan.price}</option>`
        ).join("");
        
        document.getElementById("systemPlan").innerHTML = `<option value="">Select Option</option>${planOptions}`;
        $('select').material_select('destroy');
        $('select').material_select();
    }

    function validateQuantity(min, max) {
        let valid = true;
        const quantityElement = document.getElementById("quantity");
        const quantity = parseInt(quantityElement.value);
        const quantifyError = document.getElementById("quantityError");

        // Set min and max length for the quantity input
        quantityElement.setAttribute("minlength", min);
        quantityElement.setAttribute("maxlength", max);

        // Reset any previous error messages
        quantifyError.style.display = "none"; // Initially hide the error message

        if (isNaN(quantity)) {
            quantifyError.innerHTML = "Please enter a valid number.";
            valid = false;
        } else if (quantity > max) {
            quantifyError.innerHTML = `The maximum quantity allowed is ${max}.`;
            valid = false;
        } else if (quantity < min) {
            quantifyError.innerHTML = `The minimum quantity allowed is ${min}.`;
            valid = false;
        }

        // If valid, you can proceed with further logic (e.g., submitting the form or hiding the error)
        if (valid) {
            quantityElement.setCustomValidity("");
            quantifyError.style.display = "none"; // Hide error message when valid
            quantifyError.classList.remove("text-danger"); // Remove danger class
        } else {
            quantityElement.setCustomValidity(quantifyError.innerHTML);
            quantifyError.style.display = "block"; // Show error
            quantifyError.classList.add("text-danger");
        }
    }

    document.addEventListener("DOMContentLoaded", async function() {
        const amount = document.getElementById("socialAmountAvailable");

        if (amount) {

            document.forms["systemServiceForm"].action = "../buy/socially.php";
            document.getElementById("quantity").insertAdjacentHTML("afterend", `<div id="quantityInfo"></div>`);

            // Insert the category select element before the systemPlan element
            const inputWrapper = document.getElementById("systemPlan").parentElement;
            if (inputWrapper) {
                const categoryElement = document.createElement("div");
                inputWrapper.insertBefore(categoryElement, inputWrapper.firstChild);
                categoryElement.className = "category";
                categoryElement.innerHTML = `
                    <select onChange="changeCategory()" id="category" name="category">
                        <option value="">Select Category</option>
                    </select>
                `;
            }

            // Create a container for the plan description if it doesn't exist
            if (!document.getElementById("planDescription")) {
                document.getElementById("systemPlan").insertAdjacentHTML("afterend", `<div id="planDescription"></div>`);
            }

            let categories = [];
            try {
                const planRequest = await axios.get("https://gsubz.com/socially/plan/");
                if (planRequest.data && planRequest.data?.list) {
                    for (const plan of planRequest.data.list) {
                        const category = plan.category;
                        if (!planList[category]) planList[category] = [];
                        planList[category].push(plan);
                    }
                    categories = Object.keys(planList);
                    const categoryElement = document.getElementById("category");
                    categoryElement.innerHTML = `<option value="">Select Category</option>`;
                    for (const category of categories) {
                        categoryElement.innerHTML += `<option value="${category}" >${category}</option>`;
                    }

                    $('select').material_select('destroy');
                    $('select').material_select();
                    const category = document.getElementById("category");
                    category.value = categories[0];

                    changeCategory(categories[0]);
                }
            } catch (e) {
                console.error("Error fetching plans", e);
            }

            $('#systemPlan').on('change', function() {
                const planValue = document.getElementById("systemPlan").value;
                const selectedOption = document.querySelector("#systemPlan option[value='" + planValue + "']");

document.getElementById("planDescription").innerHTML = description

  : '';


                const quantityElement = document.getElementById("quantity");
                const min = Number(selectedOption.getAttribute("data-min"));
                const max = Number(selectedOption.getAttribute("data-max"));
                const price = Number(selectedOption.getAttribute("data-price"));

                document.getElementById("quantityInfo").innerHTML = `<div class="mb-3" id="quantityError"></div>
                    <div><span>Min: ${min}</span> <span>Max: ${max}</span></div><br />`;
                validateQuantity(min, max);
                const quantity = Number(quantityElement.value);
                const pricePerOne = price / 1000;
                setTimeout(function() {
                    amount.value = (pricePerOne * quantity).toFixed(2);
                }, 100);
                quantityElement.addEventListener("input", () => {
                    validateQuantity(min, max);
                    quantityElement.value = quantityElement.value.replace(/\D/g, "");
                    const newQuantity = Number(quantityElement.value);
                    const newPrice = (pricePerOne * newQuantity).toFixed(2);
                    amount.value = newPrice;
                });
            });
        }
